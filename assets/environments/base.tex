% THEOREM STYLE DEFINITIONS
% ==============================================================================

% --- Default theorem style: bold sans-serif head, centered on its line ---
\newtheoremstyle{default}{\topsep}{\topsep}%
    {}% Body font
    {}% Indent amount (empty = no indent)
    {\sffamily\bfseries}% Theorem head font
    {.}% Punctuation after theorem head
    { }% Space after theorem head
    {\thmname{#1}\thmnumber{~#2}\thmnote{~#3}}% Head specification

% --- Non-numbered theorem style ---
\newtheoremstyle{nonum}{\topsep}{\topsep}%
    {}%
    {}%
    {\sffamily\bfseries}%
    {.}%
    { }%
    {\thmname{#1}\thmnote{~#3}}%

% ==============================================================================
% \mytheorem MACRO
% ==============================================================================

% \mytheorem{name}{Display Name}{color}{counter}{type}
% Defines a left-accented theorem environment using mdframed.
% - If type is 'o', uses a standalone counter.
% - Otherwise, the counter is attached to the enclosing section.
\newcommand{\mytheorem}[5]{%
  \ifstrequal{#5}{o}{% Standalone counter
    \newmdtheoremenv[
      hidealllines=true,
      leftline=true,
      skipabove={5},
      innertopmargin={0.1},
      innerbottommargin=5pt,
      linewidth=4pt,
      linecolor=#3,
      backgroundcolor=theorembg,
      fontcolor=theoremtext,
      % linewidthcolor=theoremframe,
    ]{#1}[#4]{#2}%
  }{% Dependent counter
    \newmdtheoremenv[
      hidealllines=true,
      leftline=true,
      skipabove={5},
      innertopmargin={0.1},
      innerbottommargin=5pt,
      linewidth=4pt,
      linecolor=#3,
      backgroundcolor=theorembg,
      fontcolor=theoremtext,
      % linewidthcolor=theoremframe,
    ]{#1}{#2}[#4]%
  }%
}
